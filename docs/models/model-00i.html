  subgraph PLAN["Policy and budget formulation - multiple fiscal years"]
    P["Development and investment planning"] --> F["Fiscal framework"] --> B["Agency and executive proposals"] --> G["Enacted appropriation"]
  end
  subgraph EXP["Expenditure authority and service delivery"]
    A["Applicable allotment / budget authority"] --> O["Commitment / obligation"] --> D["Delivery, entitlement or service"] --> L["Liability / verified claim"]
  end
  subgraph CASH["Treasury and cash - continuous"]
    R["Revenue and financing"] --> T["Cash position / TSA"] --> N["Cash programming and authority"] --> PAY["Payment instruction and settlement"]
  end
  subgraph BOOK["Accounting, property and control - continuous"]
    REC["Record financial events when recognition criteria hold"] --> RECON["Reconcile budget, books, bank and assets"] --> REP["Financial and performance reports"]
  end
  subgraph OV["Assurance and feedback - throughout and afterwards"]
    IC["Agency controls and internal audit"] --> COA["COA independent audit and findings"] --> FEED["Corrective action and future plans"]
  end
