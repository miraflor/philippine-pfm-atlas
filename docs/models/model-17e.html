  %% SOURCES: Proposed conceptual model; actual agency field links not proven; GENERAL KNOWLEDGE / atlas inference

flowchart LR
  PIP["Investment project ID (if assigned)"] --> XW["Documented project-PAP crosswalk"]
  XW --> PAP["Program / activity / project code and budget version"]
  PAP --> APP["Appropriation item and fiscal year"] --> ALL["Allotment source / release ID"]
  ALL --> ORS["One or many obligation references"]
  CON["Contract or award ID"] --> ORS
  ORS --> CLAIM["One or many deliveries / billings / vouchers"]
  CLAIM --> PAY["One or many bank payment IDs"]
  CLAIM --> GL["Accounting entry / liability ID"]
  PAY --> GL
  UACS["UACS class code with effective dates"] -.-> PAP
  UACS -.-> ORS
  UACS -.-> GL
  NOTE["Links require evidence; identical UACS does not prove same event"] -.-> XW
